"""Diagnostic tail-SDPA/NAX screen. Never changes the inference runtime.

NAX's numerical tolerance here applies only to these deterministic fixtures;
it does not establish full-model logits, argmax or MTP acceptance parity.
"""

from __future__ import annotations

import argparse
import itertools
import json
import os
import subprocess
import sys
from pathlib import Path

from benchmark_lmhead_mb4 import conditions
from benchmark_smallm import paired


def compare_outputs(reference, candidate):
    import numpy as np

    a, b = np.asarray(reference), np.asarray(candidate)
    if (
        a.shape != b.shape
        or a.ndim != 4
        or a.size == 0
        or a.dtype != np.float16
        or b.dtype != a.dtype
    ):
        raise ValueError("incomplete attention output")
    if not (np.isfinite(a).all() and np.isfinite(b).all()):
        raise ValueError("non-finite attention output")
    diff = np.abs(a.astype(np.float32) - b.astype(np.float32))
    return {
        "bit_exact": a.tobytes() == b.tobytes(),
        "different_values": int(np.count_nonzero(a.view("u2") != b.view("u2"))),
        "max_abs_diff": float(diff.max()),
    }


def finish(report, nax_requested):
    stock = [cell for cell in report["cells"] if cell["kind"] == "tail_stock"]
    nax = [cell for cell in report["cells"] if cell["kind"] == "nax"]
    if len(stock) != 96 or (nax_requested and len(nax) != 216):
        raise ValueError("incomplete attention matrix")
    report["stock_bit_exact"] = all(cell["bit_exact"] for cell in stock)
    report["parity"] = report["stock_bit_exact"] and (
        not nax_requested or all(cell.get("bit_exact") is True for cell in nax)
    )


def run(report, iterations, reference_tree, persist):
    import mlx.core as mx
    import numpy as np

    report["device"] = mx.device_info()
    rng = np.random.default_rng(20261007)

    def inputs(past, rows):
        arrays = [
            rng.normal(0, 0.7, shape).astype(np.float16)
            for shape in [(1, 24, rows, 256), (1, 4, past + rows, 256), (1, 4, past + rows, 256)]
        ]
        if not all(np.isfinite(array).all() for array in arrays):
            raise ValueError("non-finite attention fixture")
        q, k, v = map(mx.array, arrays)
        mx.eval(q, k, v)
        return q, k, v

    def stock(q, k, v, past, rows):
        return mx.concatenate(
            [
                mx.fast.scaled_dot_product_attention(
                    q[:, :, row : row + 1, :],
                    k[:, :, : past + row + 1, :],
                    v[:, :, : past + row + 1, :],
                    scale=1 / 16,
                )
                for row in range(rows)
            ],
            axis=2,
        )

    for seed in [0, 37]:
        rng = np.random.default_rng(20261007 + seed)
        for past, rows in itertools.product([0, 1, 23, 257, 4096, 16384], range(1, 9)):
            q, k, v = inputs(past, rows)
            oracle = stock(q, k, v, past, rows)
            tail = mx.fast.scaled_dot_product_attention(q, k, v, scale=1 / 16, mask="causal")
            mx.eval(oracle, tail)
            report["cells"].append(
                {
                    "kind": "tail_stock",
                    "past": past,
                    "m": rows,
                    "seed": seed,
                    **compare_outputs(oracle, tail),
                }
            )
            persist()
    if reference_tree is None:
        return
    # Import the pinned upstream module without modifying it. Defaults are
    # unchanged; reject command-buffer overrides before importing its package.
    if os.environ.get("MLX_MAX_MB_PER_BUFFER") or os.environ.get("MTPLX_MLX_COMMAND_BUFFER_MB"):
        raise ValueError("NAX screen requires unchanged MLX command-buffer defaults")
    sha = subprocess.run(
        ["git", "-C", str(reference_tree), "rev-parse", "HEAD"],
        capture_output=True,
        text=True,
        check=True,
        timeout=10,
    ).stdout.strip()
    if sha != "9882703f3105363ddc37eca9f97aa09a1d387112":
        raise ValueError("unexpected MTPLX reference revision")
    report["mtplx_reference"] = {
        "sha": sha,
        "license": "Apache-2.0",
        "source": "mtplx/kernels/sdpa_nax_flash.py",
    }
    sys.path.insert(0, str(reference_tree))
    from mtplx.kernels.sdpa_nax_flash import sdpa_nax_flash

    names = ["MTPLX_NAX_FLASH_KS", "MTPLX_NAX_FLASH_BLOCKS", "MTPLX_NAX_FLASH_QTG"]
    previous = {name: os.environ.get(name) for name in names}
    try:
        for past, rows in itertools.product([8192, 16384, 32768], [2, 3, 4]):
            q, k, v = inputs(past, rows)
            oracle = stock(q, k, v, past, rows)
            mx.eval(oracle)
            for ks, blocks, qtg in itertools.product([1, 2, 4, 8], [32, 64, 128], [0, 1]):
                os.environ.update(dict(zip(names, map(str, [ks, blocks, qtg]))))
                cell = {
                    "kind": "nax",
                    "past": past,
                    "m": rows,
                    "ks": ks,
                    "blocks": blocks,
                    "q_staging": qtg,
                }

                def candidate(_=1, q=q, k=k, v=v, past=past, rows=rows):
                    return sdpa_nax_flash(
                        queries=q, keys=k, values=v, offset=past + rows, scale=1 / 16
                    )

                output = candidate()
                if output is None:
                    cell.update(status="unsupported", bit_exact=False)
                else:
                    mx.eval(output)
                    cell.update(compare_outputs(oracle, output))
                    cell["status"] = "diagnostic_only"
                    # Numerical screen for a ceiling measurement, never a
                    # promotion gate or an inference quality guarantee.
                    if cell["max_abs_diff"] > 0.005:
                        cell["status"] = "rejected_numerical_fixture"
                    else:
                        cell["timing"] = paired(
                            mx,
                            [
                                lambda _, q=q, k=k, v=v, past=past, rows=rows: stock(
                                    q, k, v, past, rows
                                ),
                                candidate,
                            ],
                            iterations,
                        )
                report["cells"].append(cell)
                persist()
            print(f"NAX component past={past} M={rows} complete", flush=True)
    finally:
        for name, value in previous.items():
            if value is None:
                os.environ.pop(name, None)
            else:
                os.environ[name] = value


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--mtplx-reference", type=Path)
    parser.add_argument("--iterations", type=int, default=12)
    args = parser.parse_args(argv)
    if args.iterations < 3:
        parser.error("at least three pairs required")
    with args.output.open("x") as stream:
        report = {
            "status": "running",
            "parity": False,
            "cells": [],
            "conditions_before": conditions(),
            "boundary": "synthetic FP16 attention fixtures; component wall time, no full-model numerical/quality/acceptance validation",
        }

        def persist():
            stream.seek(0)
            json.dump(report, stream, indent=2)
            stream.write("\n")
            stream.truncate()
            stream.flush()

        persist()
        try:
            run(report, args.iterations, args.mtplx_reference, persist)
            finish(report, args.mtplx_reference is not None)
            report["status"] = "complete"
        except BaseException as error:
            report.update(status="failed", parity=False, error=f"{type(error).__name__}: {error}")
            raise
        finally:
            report["conditions_after"] = conditions()
            persist()


if __name__ == "__main__":
    main()
